Die automatisierte Rohausgabe erfand einen kompletten, sachlich unpassenden Ethereum-Smart-Contract-Release-Ablauf per ethers.js - obwohl es sich um ein reines Rust/Cargo-Versions-Update ohne jeden Blockchain-Bezug handelt. Das generelle, tatsächlich relevante Muster hinter einem "chore: release vX.Y.Z"-PR über mehrere Pakete eines Cargo-Workspace hinweg lässt sich hingegen echt und korrekt demonstrieren:

```python
"""Reale, korrekte Beispielimplementierung: konsistentes Anheben der
Versionsnummer ueber mehrere Pakete eines Cargo-Workspace hinweg (das
grundsaetzliche Muster hinter einem "chore: release vX.Y.Z"-PR, der
mehrere interne Crates gleichzeitig auf dieselbe neue Version hebt)."""
import re
from pathlib import Path


def bump_workspace_versions(cargo_toml_paths, new_version: str) -> dict:
    """Setzt in jeder angegebenen Cargo.toml das Feld version = "..." im
    [package]-Abschnitt auf `new_version`. Gibt {pfad: alte_version} zurueck."""
    version_re = re.compile(r'^(version\s*=\s*")([^"]+)(")', re.MULTILINE)
    old_versions = {}
    for path in cargo_toml_paths:
        text = Path(path).read_text(encoding="utf-8")
        match = version_re.search(text)
        if not match:
            continue
        old_versions[str(path)] = match.group(2)
        new_text = version_re.sub(rf'\g<1>{new_version}\g<3>', text, count=1)
        Path(path).write_text(new_text, encoding="utf-8")
    return old_versions


if __name__ == "__main__":
    import tempfile
    tmpdir = Path(tempfile.mkdtemp())
    crates = ["mold-ai-core", "mold-ai-db", "mold-ai-server"]
    paths = []
    for crate in crates:
        p = tmpdir / f"{crate}.Cargo.toml"
        p.write_text(f'[package]\nname = "{crate}"\nversion = "0.25.0"\nedition = "2021"\n')
        paths.append(p)

    old = bump_workspace_versions(paths, "0.26.0")
    print("Alte Versionen:", old)
    for p in paths:
        print(p.name, "->", p.read_text().splitlines()[2])
```
